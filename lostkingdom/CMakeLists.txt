# LOST KINGDOM — AGR Engine build (Android NDK shared library + Linux desktop test harness)
cmake_minimum_required(VERSION 3.22)
project(LostKingdom CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB ENGINE_SOURCES
    engine/gfx/*.cpp engine/anim/*.cpp engine/audio/*.cpp
    engine/platform/PlatformCommon.cpp)
file(GLOB GAME_SOURCES game/*.cpp)
if(NOT GAME_SOURCES OR LK_SMOKE)
    set(GAME_SOURCES tests/RenderSmoke.cpp)
endif()

if(ANDROID)
    add_library(native_app_glue STATIC ${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c)
    target_include_directories(native_app_glue PUBLIC ${ANDROID_NDK}/sources/android/native_app_glue)
    add_library(lostkingdom SHARED ${ENGINE_SOURCES} ${GAME_SOURCES} engine/platform/AndroidMain.cpp)
    target_compile_options(lostkingdom PRIVATE -O2 -ffast-math -fno-finite-math-only -Wall -Wno-unused-function -Wno-unused-variable -Wno-sign-compare)
    # export ANativeActivity_onCreate from the static glue
    set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -u ANativeActivity_onCreate -Wl,-z,max-page-size=16384")
    target_link_libraries(lostkingdom native_app_glue android log EGL GLESv3 aaudio m)
else()
    find_library(EGL_LIB EGL)
    find_library(GLES_LIB GLESv2)
    add_executable(lk_desktop ${ENGINE_SOURCES} ${GAME_SOURCES} engine/platform/DesktopMain.cpp)
    target_compile_options(lk_desktop PRIVATE -O2 -g -Wall -Wno-unused-function -Wno-unused-variable -Wno-sign-compare)
    target_link_libraries(lk_desktop ${EGL_LIB} ${GLES_LIB} pthread)
    file(GLOB TEST_SOURCES tests/test_*.cpp)
    if(TEST_SOURCES)
        list(FILTER GAME_SOURCES EXCLUDE REGEX ".*Main\\.cpp$")
        add_executable(lk_tests ${ENGINE_SOURCES} ${GAME_SOURCES} ${TEST_SOURCES} engine/platform/DesktopMain.cpp)
        target_compile_definitions(lk_tests PRIVATE LK_UNIT_TESTS=1)
        target_compile_options(lk_tests PRIVATE -O1 -g)
        target_link_libraries(lk_tests ${EGL_LIB} ${GLES_LIB} pthread)
    endif()
endif()
