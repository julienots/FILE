# The game player: runs exported games. On Android it is the native library (libmain.so)
# loaded by the activity; elsewhere an executable.
set(_player_libs agr::platform agr::render agr::twod agr::ui agr::script agr::animation agr::physics2d agr::physics3d agr::audio)
if(ANDROID)
    add_library(agr_player SHARED main.cpp)
    set_target_properties(agr_player PROPERTIES OUTPUT_NAME main)
    target_link_libraries(agr_player PRIVATE ${_player_libs} agr::compiler_options android log)
else()
    add_executable(agr_player main.cpp)
    target_link_libraries(agr_player PRIVATE ${_player_libs} agr::compiler_options)
endif()
if(TARGET SDL3::SDL3-static)
    target_link_libraries(agr_player PRIVATE SDL3::SDL3-static)
else()
    target_link_libraries(agr_player PRIVATE SDL3::SDL3)
endif()
set_target_properties(agr_player PROPERTIES FOLDER "Apps")
